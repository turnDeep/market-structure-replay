def load(path): return json.loads(Path(path).read_text(encoding='utf-8-sig'))
def jl(name): return [json.loads(line) for line in (BASE/(name+'.jsonl')).read_text(encoding='utf-8').splitlines() if line]
def save(name,value): (OUT/name).write_text(json.dumps(value,ensure_ascii=False,indent=2,allow_nan=False),encoding='utf-8')

assert all(meta['campaign_config'][k]==v for k,v in baseline['campaign_config'].items())

def mark(code,label,i,p,known):
    assert 0<=i<=known<len(bars)
    return dict(code=code,label=label,index=i,date=day(i),price=p,known_index=known)
def level(key,label,p,i): return dict(id=key,label=label,price=p,from_index=i)
def step(code,title,detail,passed=True): return dict(code=code,title=title,detail=detail,passed=passed)
def visual(family,i,mechanism,focus,marks,levels,steps,adopted=None,gates=None):
    assert all(m['known_index']<=i for m in marks)
    assert all(l['from_index']<=i for l in levels)
    return dict(schema=1,family=family,known_index=i,decision_index=i,date=day(i),
        mechanism=mechanism,focus_start=focus,adopted_zone=adopted,marks=marks,levels=levels,steps=steps,
        gates=gates or [],fvg_required=False,sma_required=False,
        execution_step=step('4','翌始値で執行','この判断日時点で、翌始値・約定結果は未観測。',None))

decisions=[]
for oid,c in contexts.items():
    i=c['decision_index'];b=bars[i];role=c['role'];th=c['reentry_entry_thesis']
    assert all(v['passed'] for v in c['close_gate_snapshot'].values()),oid
    gates=[dict(name=k,passed=v['passed'],status=v['status'],reason=v['reason']) for k,v in c['close_gate_snapshot'].items()]

        marks=[mark('1','売却後の安値',low,th['reaction_low'],role['known_index']),
               mark('H','回復参照高値',hi,th['structure_recovery_level'],role['known_index']),
               mark('2','構造突破',role['breakout_index'],bars[role['breakout_index']]['close'],role['known_index']),
               mark('3','回復BUY判断',i,b['close'],i)]
        levels=[level('recovery-ref','回復参照高値',th['structure_recovery_level'],role['known_index']),
                level('recovery-floor','回復・執行下限',threshold,i),level('sl','初期SL',stop,i)]
        steps=[step('1','売却後の新しい安値',f"前回実決済 {day(q['cutoff'])} の後、{day(low)} に構造安値 {fmt(th['reaction_low'])}。OB接触は不要。"),
               step('2','回復構造を確認',f"参照高値 {fmt(th['structure_recovery_level'])}。突破 {day(role['breakout_index'])}、認識 {day(role['known_index'])}、有効 {day(role['effective_index'])}。"),
               step('3','必要水準を回復',f"判断終値 {fmt(b['close'])} > 執行下限 {fmt(threshold)}。翌始値も下限を厳密に上回る必要がある。"),
               step('SL','構造安値から損切り設定',f"初期SL {fmt(stop)}。OB/FVGを後付けした根拠ではない。")]

    else:
        family='reversal';z=c['source_zone'];contact=c['contact_index'];roletext=roles.get(role['kind'],role['kind'])

        selected=selected_entries[i]
        early=selected['route']=='early'
        route='支持OB反転BUY' if early else '支持OB・構造確認BUY'
        if early:
            assert b['close']>=z['body_high'] and b['close']>bars[i-1]['high']
            if role['kind']=='upward_leg_support': assert b['close']>b['open']
        adopted=dict(id=z['id'],low=z['wick_low'],high=z['wick_high'],body_low=z['body_low'],body_high=z['body_high'],
            available_index=z['available_index'],origin_index=z['origin_index'],available_date=day(z['available_index']),
            origin_date=day(z['origin_index']),timeframe=z['timeframe'],role=roletext,
            role_known=day(role['known_index']),role_effective=day(role['effective_index']),
            state_at_decision='tradeable_saved_gate',decision_index=i)
        marks=[mark('2','OB接触',contact,max(z['wick_low'],min(bars[contact]['low'],z['wick_high'])),i),
               mark('L','反応安値',low,th['reaction_low'],i),mark('3','BUY判断',i,b['close'],i)]
        levels=[level('ob-body','OB実体上端',z['body_high'],contact),level('sl','初期SL',stop,i)]
        condition=f"終値 {fmt(b['close'])} ≥ 実体上端 {fmt(z['body_high'])}、前日高値 {fmt(bars[i-1]['high'])} 超え。"
        if early: levels.insert(1,level('prior-high','前日高値',bars[i-1]['high'],i-1))
        else:
            candidate=plans[i]['intent']['candidate'];ref=candidate['reference']
            boundary=ref['price']+meta['legacy_rules']['structure_buffer_atr']*ref['atr_at_origin']
            assert b['close']>boundary
            marks.append(mark('H','構造参照高値',ref['origin_index'],ref['price'],ref['available_index']))
            levels.append(level('buy-boundary','構造突破の境界',boundary,ref['available_index']))
            if candidate['confirmation']=='close_sequence':
                count=meta['legacy_rules']['buy_confirm_closes']
                assert all(bars[j]['close']>boundary for j in range(i-count+1,i+1))
                assert b['close']-candidate['low']>=meta['legacy_rules']['buy_departure_atr']*candidate['low_atr']
                condition=f"参照高値 {fmt(ref['price'])} + {meta['legacy_rules']['structure_buffer_atr']}×ATR {fmt(ref['atr_at_origin'])} = {fmt(boundary)} を{count}終値連続で上回る。安値からの離脱幅も既存閾値以上。"

            elif candidate['confirmation']=='breakout_retest':

                assert tz is not None,'OB retest adapter needs explicit shape mapping'

                assert b['close']>b['open'] and b['low']>=candidate['low']
                retest=dict(id=target,kind=tz['kind'].upper(),low=tz['low'],high=tz['high'],
                    origin_index=tz['origin_index'],available_index=tz['available_index'],
                    origin_date=day(tz['origin_index']),available_date=day(tz['available_index']),
                    decision_index=i,known_index=i,timeframe='1D')
                levels.extend([level('retest-low','採用FVG下端',tz['low'],tz['available_index']),
                               level('retest-high','採用FVG上端',tz['high'],tz['available_index'])])
                marks.append(mark('R','採用FVG再テスト',i,max(tz['low'],min(b['low'],tz['high'])),i))
                condition=f"構造確認後、{day(tz['available_index'])} に利用可能となったFVG {fmt(tz['low'])}–{fmt(tz['high'])} を再テスト。陽線でFVG上端と構造境界 {fmt(boundary)} を上回って引け、構造安値も維持。"

            else: raise AssertionError(candidate['confirmation'])
        steps=[step('1','接触前に支持役割を認識',roletext+f"。有効 {day(role['effective_index'])}、OB利用可能 {day(z['available_index'])}。"),

        focus=max(0,min(contact,low)-18);mechanism='支持役割のあるOB → 接触・反応 → 原ルートの価格条件 → 翌始値BUY'
    ev=visual(family,i,mechanism,focus,marks,levels,steps,adopted,gates)
    ev['retest_zone']=retest;ev['fvg_used']=bool(retest and retest['kind']=='FVG')
    if retest:
        ev['mechanism']='支持OBで構造回復 → 確認後のFVG再テスト → 翌始値BUY'
    lines=[s['title']+'：'+s['detail'] for s in steps]+['200SMAは参照。'+('このBUYは構造後のFVG再テストを採用。' if retest else 'このBUYではFVGを必須条件として採用していない。'),
        f'終値判断の翌有効観測日始値+{slip:g}ptでBUYするモデル。']

        zone_id=adopted['id'] if adopted else None,initial_stop=stop,contact_date=day(c['contact_index']),

        filters=['NKD=F CMEドル建て / 2026-10-14以前の有効配信日足 / 選択日・表示窓に連動'],
        metricDefinitions=[dict(label=label,definition=definition)],caveats=caveats or []))
snapshot.update(title='日経先物 NKD=F · v0.11 追体験',buildStatus='updating',generatedAt='2026-10-15')

    strategyStatus='NOT_VALIDATED',revision='rev11_without_continuation',pointValue=5,commissionSide=0,slippagePoints=5,
    fvgRole='200SMAは参照。FVGは全BUYの必須条件ではないが、構造後の再テストBUYに採用されうる。継続BUYはオフ。')

    market_prices=query(bars,'NKD=F 日足・200SMA',['NKD_F/daily_analysis.csv'],'1行=有効配信日足。200SMA=当日を含む直近200有効終値の平均。',['不正OHLC等205行は明示除外、価格補間なし。CMEドル建て連続先物で大阪先物ではない。']),
    primary_zones=query(zones,'v09 OB/FVGの状態',['primary_zone_state_events.jsonl'],'選択日以下の最終状態を採用。起点日と利用可能日は別。',['機械的な価格形状候補で機関の注文観測ではない。欠落日足の隣接関係は未補正。']),
